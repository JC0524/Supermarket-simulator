    function playTone(freq,dur,type,vol,delay){
        if(!audioCtx||S.muted)return;
        try{const t0=audioCtx.currentTime+(delay||0);const o=audioCtx.createOscillator(),g=audioCtx.createGain();
        o.connect(g);g.connect(audioCtx.destination);o.frequency.setValueAtTime(freq,t0);o.type=type||'sine';
        g.gain.setValueAtTime(0,t0);g.gain.linearRampToValueAtTime(vol||0.08,t0+0.01);
        g.gain.exponentialRampToValueAtTime(0.001,t0+dur);o.start(t0);o.stop(t0+dur);}catch(e){}
    }
    function playNoise(dur,vol,delay){
        if(!audioCtx||S.muted)return;
        try{const t0=audioCtx.currentTime+(delay||0);const len=Math.floor(audioCtx.sampleRate*dur);
        const buf=audioCtx.createBuffer(1,len,audioCtx.sampleRate);const d=buf.getChannelData(0);
        for(let i=0;i<len;i++)d[i]=(Math.random()*2-1)*0.5;
        const src=audioCtx.createBufferSource(),g=audioCtx.createGain(),flt=audioCtx.createBiquadFilter();
        src.buffer=buf;flt.type='highpass';flt.frequency.setValueAtTime(3000,t0);
        src.connect(flt);flt.connect(g);g.connect(audioCtx.destination);
        g.gain.setValueAtTime(vol||0.05,t0);g.gain.exponentialRampToValueAtTime(0.001,t0+dur);
        src.start(t0);src.stop(t0+dur);}catch(e){}
    }
    function playSweep(f1,f2,dur,type,vol,delay){
        if(!audioCtx||S.muted)return;
        try{const t0=audioCtx.currentTime+(delay||0);const o=audioCtx.createOscillator(),g=audioCtx.createGain();
        o.connect(g);g.connect(audioCtx.destination);o.type=type||'sine';
        o.frequency.setValueAtTime(f1,t0);o.frequency.exponentialRampToValueAtTime(f2,t0+dur);
        g.gain.setValueAtTime(vol||0.06,t0);g.gain.exponentialRampToValueAtTime(0.001,t0+dur);
        o.start(t0);o.stop(t0+dur);}catch(e){}
    }

    function generateStore(){
        S.store=[]; let uid=1; const oKeys=Object.keys(ORIGINS);

        S.shopList.forEach(li=>{
            const base=BASE_ITEMS.find(b=>b.id===li.baseId);
            const nv=Math.min(S.cfg.variants,oKeys.length);
            shuffle(oKeys).slice(0,nv).forEach(ok=>{S.store.push(makeItem(uid++,base,ok,true));});
        });

        const nonList=shuffle(BASE_ITEMS.filter(b=>!S.shopList.some(li=>li.baseId===b.id)));
        let rem=S.cfg.storeSize-S.store.length;
        for(const base of nonList){
            if(rem<=0)break;
            const nv=Math.min(clamp(Math.ceil(Math.random()*2),1,2),rem);
            shuffle(oKeys).slice(0,nv).forEach(ok=>{if(rem<=0)return;S.store.push(makeItem(uid++,base,ok,false));rem--;});
        }
        S.store=shuffle(S.store);
    }

    function makeItem(id,base,originKey,isListItem){
        const origin=ORIGINS[originKey];
        const pi=Math.floor(Math.random()*base.packs.length), pack=base.packs[pi];
        let unitP=base.pBase*origin.priceMul*rand(0.82,1.18)*rand(0.9,1.1);
        let totalPrice=base.mType==='c'?Math.round(unitP*pack.s):Math.round(unitP*pack.s/100);
        totalPrice=Math.max(3,totalPrice);

        let freshness=null;
        if(S.cfg.fresh&&base.per){
            const r=Math.random();
            if(r<0.12) freshness='nearExpiry'; else if(r<0.3) freshness='expiring'; else freshness='fresh';
            if(freshness!=='fresh') totalPrice=Math.max(3,Math.round(totalPrice*(1-FRESHNESS[freshness].disc)));
        }

        const displayUnit=base.mType==='c'?totalPrice/pack.s:(totalPrice/pack.s)*100;

        let fakeDisc=false,fakeOriginal=0,discLabelIdx=0;
        if(S.cfg.traps&&Math.random()<S.cfg.trapChance){
            fakeDisc=true; fakeOriginal=Math.round(totalPrice*rand(1.15,1.4));
            discLabelIdx=Math.floor(Math.random()*6);
        }

        return {
            id,baseId:base.id,emoji:base.emoji,cat:base.cat,originKey,
            quality:origin.quality-(freshness?FRESHNESS[freshness].qPen:0),
            packSize:pack.s,sizeLabel:pack.l,mType:base.mType,uLabel:base.uLabel,
            price:totalPrice,unitPrice:displayUnit,
            perishable:base.per,freshness,
            fakeDisc,fakeOriginal,discLabelIdx,
            isListItem
        };
    }

    }

    function applyCoupons(){
        let savings=0; S.coupons.forEach(c=>c.used=false);
        [...S.coupons].sort((a,b)=>b.disc-a.disc).forEach(c=>{
            if(c.type==='cat'){
                const el=S.cart.map(id=>S.store.find(i=>i.id===id)).filter(i=>i&&i.cat===c.cat);
                if(el.length){savings+=Math.min(c.disc,el[0].price);c.used=true;}
            } else if(c.type==='total'&&S.spent>=c.thresh){savings+=c.disc;c.used=true;}
        });
        return savings;
    }

    function calcScores(){
        const sc={};
        const eff=S.spent-S.couponSavings, ratio=eff/S.budget;
        if(ratio<=0.6)sc.budget=5;else if(ratio<=0.72)sc.budget=4;else if(ratio<=0.84)sc.budget=3;else if(ratio<=0.94)sc.budget=2;else sc.budget=1;

        let qS=0,qC=0;
        S.cart.forEach(id=>{const it=S.store.find(i=>i.id===id);if(it){qS+=Math.max(1,it.quality);qC++;}});
        sc.quality=clamp(Math.round((qC?qS/qC:0)*1.25),1,5);

        let tA=0,tF=0;
        S.cart.forEach(id=>{const it=S.store.find(i=>i.id===id);if(!it||!it.fakeDisc)return;const sb=S.store.filter(si=>si.baseId===it.baseId);if(sb.some(si=>si.id!==it.id&&si.unitPrice<it.unitPrice*0.95)){tA++;tF++;}});
        const cbi=new Set(S.cart.map(id=>S.store.find(i=>i.id===id)?.baseId));
        S.store.forEach(si=>{if(!si.fakeDisc||!cbi.has(si.baseId)||S.cart.includes(si.id))return;if(S.store.filter(x=>x.baseId===si.baseId).some(x=>x.id!==si.id&&x.unitPrice<si.unitPrice*0.95))tA++;});
        sc.traps=tA>0?clamp(Math.round(((tA-tF)/tA)*5),1,5):5;

        const nl=S.cart.filter(id=>{const it=S.store.find(i=>i.id===id);return it&&!it.isListItem;}).length;
        sc.discipline=clamp(5-nl-S.impulseAccepted,1,5);

        let sS=0,sC=0;
        S.cart.forEach(id=>{const it=S.store.find(i=>i.id===id);if(!it)return;const sb=S.store.filter(si=>si.baseId===it.baseId);if(sb.length>1){sS+=Math.min(...sb.map(si=>si.unitPrice))/it.unitPrice;sC++;}});
        sc.smart=sC>0?clamp(Math.round((sS/sC)*5),1,5):5;

        sc.overall=Math.round(((sc.budget+sc.quality+sc.traps+sc.discipline+sc.smart)/5)*10)/10;
        return sc;
    }

    }

    function genTips(){

    }

    /* — SIZE LABEL helper — */
    function sizeLabel(it){
        const base=BASE_ITEMS.find(b=>b.id===it.baseId);
        if(!base)return it.sizeLabel;
        if(base.mType==='w'||base.mType==='v') return it.sizeLabel; // 500g, 1L etc – universal
        // count items: append translated unit
        const unitMap={'/隻':S.lang==='en'?' pcs':' 隻','/個':S.lang==='en'?' pcs':' 個','/盒':S.lang==='en'?' boxes':' 盒','/包':S.lang==='en'?' pks':' 包','/條':S.lang==='en'?' pcs':' 條'};
        return it.sizeLabel+(unitMap[base.uLabel]||'');
    }

            <div class="card p-4 mb-4 slide-up-enter" style="animation-delay:.4s"><h3 class="font-extrabold mb-2">${t('shoppingTips')}</h3><ul class="space-y-1">${tipsH}</ul></div>
            <div class="text-center slide-up-enter" style="animation-delay:.5s"><button class="px-8 py-3 rounded-xl font-black text-lg" style="background:var(--primary);color:#fff" data-action="play-again">${t('backHome')}</button></div>
        </div>`;
    }

    /* ════════════════════════════════════
       EVENT HANDLING
    ════════════════════════════════════ */
    function bindEvents(){
        const app=document.getElementById('app');
        app.removeEventListener('click',handleClick);
        app.addEventListener('click',handleClick);
        const sel=app.querySelector('[data-action="sort"]');
        if(sel){sel.removeEventListener('change',handleSort);sel.addEventListener('change',handleSort);}
        bindImpulseEvents();
    }
    function bindImpulseEvents(){
        document.querySelectorAll('[data-action="accept-impulse"]').forEach(el=>{el.onclick=e=>{e.stopPropagation();acceptImpulse();};});
        document.querySelectorAll('[data-action="reject-impulse"]').forEach(el=>{el.onclick=e=>{e.stopPropagation();rejectImpulse();};});
    }

    function handleClick(e){
        const btn=e.target.closest('[data-action]');if(!btn)return;
        resumeAudio();
        const a=btn.dataset.action;
        switch(a){
            case 'start': sfxClick(); startGame(btn.dataset.diff); break;
            case 'add-to-cart': addToCart(parseInt(btn.dataset.itemId)); break;
            case 'remove': e.stopPropagation(); removeFromCart(parseInt(btn.dataset.itemId)); break;
            case 'clear-cart': sfxClearCart(); clearCart(); toast(t('cartCleared'),'info'); break;
            case 'checkout': sfxClick(); checkout(); break;
            case 'open-cart': sfxCartOpen(); S.cartOpen=true; render(); break;
            case 'close-cart': sfxCartClose(); S.cartOpen=false; render(); break;
            case 'filter': S.filter=btn.dataset.cat; sfxClick(); render(); break;
            case 'back-home': sfxClick(); stopImpulse(); S.screen='welcome'; render(); break;
            case 'play-again': sfxClick(); S.screen='welcome'; render(); break;
            case 'set-lang': S.lang=btn.dataset.lang; sfxClick(); render(); break;
        }
    }
    function handleSort(e){S.sortBy=e.target.value;render();}

    /* ════════════════════════════════════
       INIT
    ════════════════════════════════════ */
    initAudio();
    render();
